Build a complete Android expense tracking app called "The Mindful Spender" using Kotlin and Jetpack Compose. This app AUTOMATICALLY tracks expenses by monitoring payment notifications from UPI apps and bank apps, with the primary focus on seamless, zero-effort expense logging.


=== CORE PHILOSOPHY ===
The app's main value is AUTOMATIC expense tracking. Users should never need to manually enter expenses in normal usage. Manual entry is only a backup feature for cash payments or when notifications aren't detected.

```kotlin
   @Service
   class ExpenseTrackerService : NotificationListenerService() {
       
       override fun onNotificationPosted(sbn: StatusBarNotification) {
           val packageName = sbn.packageName
           if (packageName !in PAYMENT_APPS) return
           
           val notification = sbn.notification
           val text = notification.extras.getString(Notification.EXTRA_TEXT) ?: return
           val title = notification.extras.getString(Notification.EXTRA_TITLE) ?: ""
           
           // Parse and auto-log
           parseAndLogExpense(title, text, packageName, sbn.postTime)
       }
       
       private fun parseAndLogExpense(title: String, text: String, app: String, timestamp: Long) {
           // Extract amount, merchant
           val expense = extractExpenseData(text, app) ?: return
           
           // Check for duplicate (same amount + merchant within 5 seconds)
           if (isDuplicate(expense, timestamp)) return
           
           // Auto-categorize
           expense.category = autoCategorizeMerchant(expense.merchant)
           
           // Save automatically
           saveExpenseToDatabase(expense)
           
           // Show confirmation notification
           showAutoLoggedNotification(expense)
           
           // Check budget and warn if needed
           checkBudgetAndWarn(expense)
       }
   }
```

1. ExpenseParser.kt
```kotlin
object ExpenseParser {
    private val patterns = mapOf(
        "com.phonepe.app" to listOf(
            "Payment of Rs\\.([\\d,]+(?:\\.\\d{2})?) to (.+?) is successful".toRegex(),
            "You paid Rs\\.([\\d,]+) to (.+?)".toRegex()
        ),
        "com.google.android.apps.nbu.paisa.user" to listOf(
            "You paid ₹([\\d,]+(?:\\.\\d{2})?) to (.+)".toRegex(),
            "Sent ₹([\\d,]+) to (.+?)".toRegex()
        )
        // ... more patterns
    )
    
    fun parseNotification(title: String, text: String, packageName: String): ParsedExpense? {
        val appPatterns = patterns[packageName] ?: return null
        
        for (pattern in appPatterns) {
            val match = pattern.find(text) ?: continue
            val amount = match.groupValues[1].replace(",", "").toDoubleOrNull() ?: continue
            val merchant = match.groupValues.getOrNull(2)?.trim() ?: "Unknown"
            
            return ParsedExpense(
                amount = amount,
                merchant = cleanMerchantName(merchant),
                paymentMethod = getAppName(packageName),
                timestamp = System.currentTimeMillis()
            )
        }
        
        return null
    }
    
    private fun cleanMerchantName(raw: String): String {
        return raw
            .replace("UPI-", "")
            .replace(Regex("@[a-zA-Z]+"), "") // Remove @paytm, @ybl
            .replace(Regex("\\d{10}"), "") // Remove phone numbers
            .trim()
    }
}
```

```kotlin
class ExpenseRepository(
    private val expenseDao: ExpenseDao,
    private val budgetDao: BudgetDao
) {
    fun autoLogExpense(expense: Expense): Result<Long> {
        // Check for duplicates
        val recent = expenseDao.getExpensesInTimeWindow(
            expense.timestamp - 5000,
            expense.timestamp + 5000
        )
        
        if (recent.any { it.amount == expense.amount && it.merchant == expense.merchant }) {
            return Result.failure(Exception("Duplicate transaction"))
        }
        
        // Save to database
        val id = expenseDao.insert(expense)
        
        // Update daily total
        updateDailyTotal(expense.amount)
        
        return Result.success(id)
    }
    
    fun getTodayTotal(): Flow<Double> {
        return expenseDao.getTodayExpensesFlow()
            .map { expenses -> expenses.sumOf { it.amount } }
    }
}
```
